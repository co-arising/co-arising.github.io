package example.bankwealth.infrastructure.repositories.projection

import scalapb.GeneratedMessage
import example.bankwealth.application.readmodels.bank_journal.repository.BankJournalReadsideRepository.{Row, State}

object BankJournalReadsideDomain {
  def accountState(row: Row): example.bankwealth.domain.entities.CreditBalanceEntity.State = {
    require(row.kind == "account", "stream kind mismatch")
    row.payload.map(bytes => example.bankwealth.infrastructure.persistence.converters.DomainConverters.decodeCreditBalanceStatePO(example.bankwealth.infrastructure.proto.CreditBalanceStatePO.parseFrom(bytes)))
      .getOrElse(example.bankwealth.domain.entities.CreditBalanceEntity.empty.copy(id = example.bankwealth.domain.values.AccountId(row.resource)))
  }
  def orderState(row: Row): example.bankwealth.domain.entities.FundOrderEntity.State = {
    require(row.kind == "order", "stream kind mismatch")
    row.payload.map(bytes => example.bankwealth.infrastructure.persistence.converters.DomainConverters.decodeFundOrderStatePO(example.bankwealth.infrastructure.proto.FundOrderStatePO.parseFrom(bytes)))
      .getOrElse(example.bankwealth.domain.entities.FundOrderEntity.empty.copy(id = example.bankwealth.domain.values.FundOrderId(row.resource)))
  }
  def planState(row: Row): example.bankwealth.domain.entities.SipPlanEntity.State = {
    require(row.kind == "plan", "stream kind mismatch")
    row.payload.map(bytes => example.bankwealth.infrastructure.persistence.converters.DomainConverters.decodeSipPlanStatePO(example.bankwealth.infrastructure.proto.SipPlanStatePO.parseFrom(bytes)))
      .getOrElse(example.bankwealth.domain.entities.SipPlanEntity.empty.copy(id = example.bankwealth.domain.values.PlanId(row.resource)))
  }
  trait Ownership {
    def account(row: Row, next: example.bankwealth.domain.entities.CreditBalanceEntity.State): (Option[String], Option[String])
    def order(row: Row, next: example.bankwealth.domain.entities.FundOrderEntity.State): (Option[String], Option[String])
    def plan(row: Row, next: example.bankwealth.domain.entities.SipPlanEntity.State): (Option[String], Option[String])
  }
  def fold(row: Row, event: Any, ownership: Ownership): State = row.kind match {
    case "account" =>
      val message = event match {
        case pb: GeneratedMessage => pb
        case _ => throw new IllegalArgumentException("domain journal must contain protobuf")
      }
      val next = (new example.bankwealth.infrastructure.persistence.CreditBalanceEventAdapter)
        .fromJournal(message.asInstanceOf[example.bankwealth.infrastructure.proto.CreditBalanceEventPO], message.getClass.getSimpleName).events.foldLeft(accountState(row)) { (state, event) =>
          event match {
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.AccountInitialized =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.ReserveFundsConfirmed =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.DeductFundsConfirmed =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.ReleaseReservedFundsConfirmed =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.RecordIncomingCreditsConfirmed =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.CommitIncomingCreditsConfirmed =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case value: example.bankwealth.domain.entities.CreditBalanceEntity.CancelIncomingCreditConfirmed =>
              require(value.resource.value == row.resource, "account event identity mismatch")
            case _ => ()
          }
          example.bankwealth.domain.entities.behaviors.CreditBalanceEventHandler(state, event)
        }
      require(next.id.value == row.resource, "account identity mismatch")
      val (account, principal) = ownership.account(row, next)
      State(Some(example.bankwealth.infrastructure.persistence.converters.DomainConverters.encodeCreditBalanceStatePO(next).toByteArray), account, principal)
    case "order" =>
      val message = event match {
        case pb: GeneratedMessage => pb
        case _ => throw new IllegalArgumentException("domain journal must contain protobuf")
      }
      val next = (new example.bankwealth.infrastructure.persistence.FundOrderEventAdapter)
        .fromJournal(message.asInstanceOf[example.bankwealth.infrastructure.proto.FundOrderEventPO], message.getClass.getSimpleName).events.foldLeft(orderState(row)) { (state, event) =>
          example.bankwealth.domain.entities.behaviors.FundOrderEventHandler(state, event)
        }
      require(next.id.value == row.resource, "order identity mismatch")
      val (account, principal) = ownership.order(row, next)
      State(Some(example.bankwealth.infrastructure.persistence.converters.DomainConverters.encodeFundOrderStatePO(next).toByteArray), account, principal)
    case "plan" =>
      val message = event match {
        case pb: GeneratedMessage => pb
        case _ => throw new IllegalArgumentException("domain journal must contain protobuf")
      }
      val next = (new example.bankwealth.infrastructure.persistence.SipPlanEventAdapter)
        .fromJournal(message.asInstanceOf[example.bankwealth.infrastructure.proto.SipPlanEventPO], message.getClass.getSimpleName).events.foldLeft(planState(row)) { (state, event) =>
          example.bankwealth.domain.entities.behaviors.SipPlanEventHandler(state, event)
        }
      require(next.id.value.isEmpty || next.id.value == row.resource, "plan identity mismatch")
      val (account, principal) = ownership.plan(row, next)
      State(Some(example.bankwealth.infrastructure.persistence.converters.DomainConverters.encodeSipPlanStatePO(next).toByteArray), account, principal)
    case "payment-saga" | "payment-host" | "payment-run" | "refund-saga" | "refund-host" | "refund-run" | "ta-host" | "ta-run" | "sip-host" | "sip-run" => State(None, None, None)
    case _ => throw new IllegalArgumentException("unrecognized persistence kind")
  }
}
