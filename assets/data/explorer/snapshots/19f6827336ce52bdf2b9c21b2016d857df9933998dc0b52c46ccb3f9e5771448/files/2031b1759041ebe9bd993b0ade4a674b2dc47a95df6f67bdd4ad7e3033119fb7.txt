package example.bankwealth.application.readmodels

import example.bankwealth.domain.values._
import example.bankwealth.interfaces.api.ApiCommands.{BankCashOriginal, BankSipCreateIntent}
import example.bankwealth.interfaces.api.ApiWire.MoneyWire

object ReadModelViews {
  final case class BankProjectionWatermark(
    persistenceId: String,
    lastSequence: Long,
    observedSequence: Option[Long],
    disposition: String
  )
  final case class BankReadMetadata(
    schemaVersion: String,
    disposition: String,
    watermarks: Vector[BankProjectionWatermark]
  )
  final case class BankAccountView(
    accountId: AccountId,
    initialized: Boolean,
    available: MoneyWire,
    reserved: MoneyWire
  )
  final case class BankCashRequestView(
    original: BankCashOriginal,
    paymentStatus: String,
    taStatus: String,
    refundStatus: String
  )
  final case class BankTaSharesView(fundCode: String, shares: TaExactText, nav: TaExactText)
  final case class BankOrderView(
    orderId: FundOrderId,
    operationId: OperationId,
    investorAccountId: AccountId,
    fundCollectionAccountId: AccountId,
    fundCode: String,
    amount: MoneyWire,
    paymentTxId: Option[TxId],
    paymentStatus: String,
    taRequestId: Option[TaRequestId],
    taStatus: String,
    taReference: Option[String],
    taShares: Option[BankTaSharesView],
    taRejectionReason: Option[String],
    refundOperationId: Option[OperationId],
    refundStatus: String
  )
  final case class BankSipRevisionView(config: SipRevisionConfig, endSlot: Option[Long])
  final case class BankSipPlanView(
    planId: PlanId,
    original: Option[BankSipCreateIntent],
    initialized: Boolean,
    scheduleSealed: Boolean,
    paused: Boolean,
    currentTick: Long,
    dueSlots: Vector[Long],
    revisions: Vector[BankSipRevisionView],
    scheduleCount: Long,
    observedCount: Long,
    startCount: Long,
    successTotal: Long,
    failureTotal: Long
  )
  final case class BankSipOccurrenceView(
    occurrence: SipOccurrence,
    slotDisposition: String,
    progress: String,
    paymentStatus: String,
    taStatus: String,
    refundStatus: String,
    firstTerminal: Option[String]
  )
  final case class BankPendingOperationView(
    operationId: OperationId,
    flow: String,
    orderId: Option[FundOrderId],
    planId: Option[PlanId],
    occurrenceId: Option[OccurrenceId],
    phase: String,
    status: String,
    recoverable: Boolean
  )
  final case class BankSipPlanPage(
    metadata: BankReadMetadata,
    plans: Vector[BankSipPlanView],
    nextAfter: Option[Long]
  )
  final case class BankSipOccurrencePage(
    metadata: BankReadMetadata,
    occurrences: Vector[BankSipOccurrenceView],
    nextAfter: Option[Long]
  )
}
