package example.bankwealth.application.readmodels.bank_journal

import akka.actor.typed.ActorSystem
import akka.actor.typed.scaladsl.adapter._
import akka.projection.eventsourced.EventEnvelope
import akka.projection.jdbc.scaladsl.JdbcHandler
import akka.serialization.{SerializationExtension, Serializers}
import example.bankwealth.application.readmodels.bank_journal.repository.BankJournalReadsideRepository
import example.bankwealth.application.readmodels.bank_journal.repository.BankJournalReadsideRepository.{Payload, Row, State}
import net.imadz.common.application.projection.ScalikeJdbcSession
import scalapb.GeneratedMessage

object BankJournalReadsideProjectionHandler {
  trait Bindings {
    def identity(pid: String): (String, String)
    def fold(row: Row, event: Any): State
    def indexes(session: BankJournalReadsideRepository.Session, row: Row): Unit
  }
}

class BankJournalReadsideProjectionHandler(
  system: ActorSystem[_],
  repository: BankJournalReadsideRepository,
  bindings: BankJournalReadsideProjectionHandler.Bindings
) extends JdbcHandler[EventEnvelope[GeneratedMessage], ScalikeJdbcSession] {
  require(system != null && repository != null && bindings != null, "complete projection handler bindings required")
  private val serialization = SerializationExtension(system.toClassic)
  private val allowedKinds: Set[String] = Set("account", "order", "plan", "payment-saga", "payment-host", "payment-run", "refund-saga", "refund-host", "refund-run", "ta-host", "ta-run", "sip-host", "sip-run")

  def validateIdentity(pid: String): (String, String) = {
    require(pid != null && pid.nonEmpty && pid.length <= 2048, "bounded persistence identity required")
    val (kind, resource) = bindings.identity(pid)
    require(allowedKinds.contains(kind), "unrecognized persistence kind")
    require(kind.length <= 32 && resource != null && resource.length <= 2048, "bounded persistence identity required")
    kind -> resource
  }

  private def serializePayload(event: Any): Payload = event match {
    case message: GeneratedMessage => Payload(message.getClass.getName, message.getClass.getSimpleName, message.toByteArray)
    case ref: AnyRef =>
      val serializer = serialization.findSerializerFor(ref)
      Payload(ref.getClass.getName, serializer.identifier.toString + ":" + Serializers.manifestFor(serializer, ref), serializer.toBinary(ref))
    case _ => throw new IllegalArgumentException("journal event must be serializable")
  }

  def handleEnvelope(pid: String, sequenceNr: Long, event: Any): Unit = {
    val (kind, resource) = validateIdentity(pid)
    repository.write { session =>
      val row = session.ensureRow(pid, kind, resource)
      val serialized = serializePayload(event)
      session.event(row.pid, sequenceNr) match {
        case Some(existing) =>
          require(existing.kind == serialized.kind && existing.manifest == serialized.manifest &&
            java.util.Arrays.equals(existing.bytes, serialized.bytes), "conflicting journal sequence/payload")
        case None =>
          require(sequenceNr == Math.addExact(row.last, 1L), "noncontiguous journal sequence")
          val state = bindings.fold(row, event)
          val next = session.append(row, sequenceNr, serialized, state)
          bindings.indexes(session, next)
      }
      ()
    }
  }

  override def process(session: ScalikeJdbcSession, envelope: EventEnvelope[GeneratedMessage]): Unit =
    handleEnvelope(envelope.persistenceId, envelope.sequenceNr, envelope.event)
}
