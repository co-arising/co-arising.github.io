slice FundOrder.ConfirmTAShares

where:
  validNav         = nav > 0.0000                                             as "本期基金净值为正"
  exactFloorMath   = shares >= 0
                     and shares * nav <= order.netAmount
                     and (shares + 1) * nav > order.netAmount                 as "确认份额精确等于净申购额向下取整除以本期净值"
  paymentSucceeded = order.paymentOutcome == "succeeded"                      as "资金扣款已成功"
  taPending        = order.taStatus != "confirmed"                            as "TA 尚未完成份额确认"

rules:
  [TA_INVALID_NAV]           IF not validNav
                             WHEN RecordFundTAConfirmed
                             THEN SHALL reject request with TA_INVALID_NAV

  [TA_SHARES_MISMATCH]       IF not exactFloorMath
                             WHEN RecordFundTAConfirmed
                             THEN SHALL reject request with TA_SHARES_MISMATCH

  [TA_PAYMENT_NOT_SUCCEEDED] IF not paymentSucceeded
                             WHEN RecordFundTAConfirmed
                             THEN SHALL reject request with TA_PAYMENT_NOT_SUCCEEDED

  [TA_ALREADY_CONFIRMED]     IF not taPending
                             WHEN RecordFundTAConfirmed
                             THEN SHALL reject request with TA_ALREADY_CONFIRMED

  [TA_CONFIRMED]             WHILE paymentSucceeded and taPending
                             WHEN RecordFundTAConfirmed AND validNav AND exactFloorMath
                             SHALL transition order.taStatus -> "confirmed"
                               AND emit FundTASharesConfirmed
