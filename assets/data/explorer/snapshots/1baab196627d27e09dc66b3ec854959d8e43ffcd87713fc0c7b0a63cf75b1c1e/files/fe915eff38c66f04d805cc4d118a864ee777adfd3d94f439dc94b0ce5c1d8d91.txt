package example.bankwealth.infrastructure.bootstrap

import scala.concurrent.Future

import example.bankwealth.application.process._
import example.bankwealth.application.process.BankRefundPorts.CompletionContext
import example.bankwealth.application.process.BankRefundMessages._
import example.bankwealth.application.process.CreditOriginalInvestorStep._
import example.bankwealth.application.process.DebitFundCollectionStep._
import net.imadz.infra.saga.strict.OperationIdentity

/** Exact bindings of the handwritten implementations to the declared ports.
  *
  * This is NOT a dependency-injection container: it is the executable form of
  * the binding obligations declared in the truth/artifact contract. The
  * composition root fills one record per port, `install` verifies every
  * declared identity and returns the `BankRefundPorts` surface — or refuses.
  * Each port role exposes exactly the one face its consumers call, so a
  * binding record carries no optional faces and no nulls. Implementation
  * identity strings keep the knowledge-side spelling; renaming members never
  * retypes them. */
object BankRefundPortBindings {

  private val implementationVersion = "bank-v1-1"

  /** One obligation row per port: the declared implementation identity and the
    * accessor to its binding. The check and the evidence maps are written
    * once, below — never per port. */
  private final case class Obligation(declaredId: String, bound: Bindings => PortBinding[_, _])

  private val obligations: Vector[Obligation] = Vector(
    Obligation("BankRefundEntry", _.entry),
    Obligation("BankRefundAssociation", _.association),
    Obligation("DebitFundCollectionPrepareExecute", _.debitFundCollectionPrepareExecute),
    Obligation("DebitFundCollectionPrepareQueryPort", _.debitFundCollectionPrepareQuery),
    Obligation("DebitFundCollectionCommitExecute", _.debitFundCollectionCommitExecute),
    Obligation("DebitFundCollectionCommitQueryPort", _.debitFundCollectionCommitQuery),
    Obligation("DebitFundCollectionCompensateExecute", _.debitFundCollectionCompensateExecute),
    Obligation("DebitFundCollectionCompensateQueryPort", _.debitFundCollectionCompensateQuery),
    Obligation("CreditOriginalInvestorPrepareExecute", _.creditOriginalInvestorPrepareExecute),
    Obligation("CreditOriginalInvestorPrepareQueryPort", _.creditOriginalInvestorPrepareQuery),
    Obligation("CreditOriginalInvestorCommitExecute", _.creditOriginalInvestorCommitExecute),
    Obligation("CreditOriginalInvestorCommitQueryPort", _.creditOriginalInvestorCommitQuery),
    Obligation("CreditOriginalInvestorCompensateExecute", _.creditOriginalInvestorCompensateExecute),
    Obligation("CreditOriginalInvestorCompensateQueryPort", _.creditOriginalInvestorCompensateQuery),
    Obligation("BankRefundSucceededAuthority", _.succeededAuthority),
    Obligation("BankRefundSucceededDelivery", _.succeededDelivery),
    Obligation("BankRefundRejectedBeforeEffectsAuthority", _.rejectedBeforeEffectsAuthority),
    Obligation("BankRefundRejectedBeforeEffectsDelivery", _.rejectedBeforeEffectsDelivery),
    Obligation("BankRefundCompensatedAuthority", _.compensatedAuthority),
    Obligation("BankRefundCompensatedDelivery", _.compensatedDelivery),
    Obligation("BankRefundRecoveryRequiredAuthority", _.recoveryRequiredAuthority),
    Obligation("BankRefundRecoveryRequiredDelivery", _.recoveryRequiredDelivery),
    Obligation("BankRefundOutcomeUnknownAuthority", _.outcomeUnknownAuthority),
    Obligation("BankRefundOutcomeUnknownDelivery", _.outcomeUnknownDelivery),
  )

  sealed trait PortBinding[I, O] {
    def id: String
    def version: String

    def requireExact(expectedId: String, expectedVersion: String): Unit = {
      require(id == expectedId, s"implementation identity mismatch: expected $expectedId, got $id")
      require(version == expectedVersion,
          s"implementation version mismatch: $id expected $expectedVersion, got $version")
    }
  }

  /** The typed implementation records. Each carries exactly one face; a null
    * face fails at construction, before install ever sees it. */
  final case class RunBinding[I, O](id: String, version: String, run: I => Future[O]) extends PortBinding[I, O] {
    require(run != null, s"implementation $id carries no run face")
  }
  final case class OperationBinding[I,
      O](id: String, version: String, run: (I, OperationIdentity) => Future[O]) extends PortBinding[I, O] {
    require(run != null, s"implementation $id carries no operation face")
  }
  final case class CompletionBinding[I,
      O](id: String, version: String, run: (I, CompletionContext) => Future[O]) extends PortBinding[I, O] {
    require(run != null, s"implementation $id carries no completion face")
  }

  /** The typed implementation record — the composition root fills exactly
    * these slots, one per declared port. */
  final case class Bindings(
    entry: RunBinding[BankRefundOriginal, Boolean],
    association: RunBinding[BankRefundOriginal, Boolean],
    debitFundCollectionPrepareExecute: OperationBinding[DebitFundCollectionPrepareInput,
        DebitFundCollectionPrepareReceipt],
    debitFundCollectionPrepareQuery: OperationBinding[DebitFundCollectionPrepareQuery,
        Option[DebitFundCollectionPrepareReceipt]],
    debitFundCollectionCommitExecute: OperationBinding[DebitFundCollectionCommitInput,DebitFundCollectionCommitReceipt],
    debitFundCollectionCommitQuery: OperationBinding[DebitFundCollectionCommitQuery,
        Option[DebitFundCollectionCommitReceipt]],
    debitFundCollectionCompensateExecute: OperationBinding[DebitFundCollectionCompensateInput,
        DebitFundCollectionCompensateReceipt],
    debitFundCollectionCompensateQuery: OperationBinding[DebitFundCollectionCompensateQuery,
        Option[DebitFundCollectionCompensateReceipt]],
    creditOriginalInvestorPrepareExecute: OperationBinding[CreditOriginalInvestorPrepareInput,
        CreditOriginalInvestorPrepareReceipt],
    creditOriginalInvestorPrepareQuery: OperationBinding[CreditOriginalInvestorPrepareQuery,
        Option[CreditOriginalInvestorPrepareReceipt]],
    creditOriginalInvestorCommitExecute: OperationBinding[CreditOriginalInvestorCommitInput,
        CreditOriginalInvestorCommitReceipt],
    creditOriginalInvestorCommitQuery: OperationBinding[CreditOriginalInvestorCommitQuery,
        Option[CreditOriginalInvestorCommitReceipt]],
    creditOriginalInvestorCompensateExecute: OperationBinding[CreditOriginalInvestorCompensateInput,
        CreditOriginalInvestorCompensateReceipt],
    creditOriginalInvestorCompensateQuery: OperationBinding[CreditOriginalInvestorCompensateQuery,
        Option[CreditOriginalInvestorCompensateReceipt]],
    succeededAuthority: CompletionBinding[BankRefundOriginal, BankRefundResult],
    succeededDelivery: CompletionBinding[BankRefundResult, Boolean],
    rejectedBeforeEffectsAuthority: CompletionBinding[BankRefundOriginal, BankRefundResult],
    rejectedBeforeEffectsDelivery: CompletionBinding[BankRefundResult, Boolean],
    compensatedAuthority: CompletionBinding[BankRefundOriginal, BankRefundResult],
    compensatedDelivery: CompletionBinding[BankRefundResult, Boolean],
    recoveryRequiredAuthority: CompletionBinding[BankRefundOriginal, BankRefundResult],
    recoveryRequiredDelivery: CompletionBinding[BankRefundResult, Boolean],
    outcomeUnknownAuthority: CompletionBinding[BankRefundOriginal, BankRefundResult],
    outcomeUnknownDelivery: CompletionBinding[BankRefundResult, Boolean],
  )

  /** Verify every declared obligation and expose the typed port surface. */
  def install(bindings: Bindings): BankRefundPorts = {
    require(bindings != null, "complete technical port bindings required")
    for (o <- obligations) o.bound(bindings).requireExact("bankwealth." + o.declaredId, implementationVersion)
    new BoundPorts(bindings)
  }

  private final class BoundPorts(bindings: Bindings) extends BankRefundPorts {
    def entry(input: BankRefundOriginal): Future[Boolean] = bindings.entry.run(input)
    def association(input: BankRefundOriginal): Future[Boolean] = bindings.association.run(input)

    def debitFundCollectionPrepareExecute(input: DebitFundCollectionPrepareInput,
        identity: OperationIdentity): Future[DebitFundCollectionPrepareReceipt] =
      bindings.debitFundCollectionPrepareExecute.run(input, identity)
    def debitFundCollectionPrepareQuery(input: DebitFundCollectionPrepareQuery,
        identity: OperationIdentity): Future[Option[DebitFundCollectionPrepareReceipt]] =
      bindings.debitFundCollectionPrepareQuery.run(input, identity)
    def debitFundCollectionCommitExecute(input: DebitFundCollectionCommitInput,
        identity: OperationIdentity): Future[DebitFundCollectionCommitReceipt] =
      bindings.debitFundCollectionCommitExecute.run(input, identity)
    def debitFundCollectionCommitQuery(input: DebitFundCollectionCommitQuery,
        identity: OperationIdentity): Future[Option[DebitFundCollectionCommitReceipt]] =
      bindings.debitFundCollectionCommitQuery.run(input, identity)
    def debitFundCollectionCompensateExecute(input: DebitFundCollectionCompensateInput,
        identity: OperationIdentity): Future[DebitFundCollectionCompensateReceipt] =
      bindings.debitFundCollectionCompensateExecute.run(input, identity)
    def debitFundCollectionCompensateQuery(input: DebitFundCollectionCompensateQuery,
        identity: OperationIdentity): Future[Option[DebitFundCollectionCompensateReceipt]] =
      bindings.debitFundCollectionCompensateQuery.run(input, identity)

    def creditOriginalInvestorPrepareExecute(input: CreditOriginalInvestorPrepareInput,
        identity: OperationIdentity): Future[CreditOriginalInvestorPrepareReceipt] =
      bindings.creditOriginalInvestorPrepareExecute.run(input, identity)
    def creditOriginalInvestorPrepareQuery(input: CreditOriginalInvestorPrepareQuery,
        identity: OperationIdentity): Future[Option[CreditOriginalInvestorPrepareReceipt]] =
      bindings.creditOriginalInvestorPrepareQuery.run(input, identity)
    def creditOriginalInvestorCommitExecute(input: CreditOriginalInvestorCommitInput,
        identity: OperationIdentity): Future[CreditOriginalInvestorCommitReceipt] =
      bindings.creditOriginalInvestorCommitExecute.run(input, identity)
    def creditOriginalInvestorCommitQuery(input: CreditOriginalInvestorCommitQuery,
        identity: OperationIdentity): Future[Option[CreditOriginalInvestorCommitReceipt]] =
      bindings.creditOriginalInvestorCommitQuery.run(input, identity)
    def creditOriginalInvestorCompensateExecute(input: CreditOriginalInvestorCompensateInput,
        identity: OperationIdentity): Future[CreditOriginalInvestorCompensateReceipt] =
      bindings.creditOriginalInvestorCompensateExecute.run(input, identity)
    def creditOriginalInvestorCompensateQuery(input: CreditOriginalInvestorCompensateQuery,
        identity: OperationIdentity): Future[Option[CreditOriginalInvestorCompensateReceipt]] =
      bindings.creditOriginalInvestorCompensateQuery.run(input, identity)

    def succeededAuthorityCompletion(input: BankRefundOriginal, context: CompletionContext): Future[BankRefundResult] =
      bindings.succeededAuthority.run(input, context)
    def succeededDeliveryCompletion(input: BankRefundResult, context: CompletionContext): Future[Boolean] =
      bindings.succeededDelivery.run(input, context)
    def rejectedBeforeEffectsAuthorityCompletion(input: BankRefundOriginal,
        context: CompletionContext): Future[BankRefundResult] =
      bindings.rejectedBeforeEffectsAuthority.run(input, context)
    def rejectedBeforeEffectsDeliveryCompletion(input: BankRefundResult, context: CompletionContext): Future[Boolean] =
      bindings.rejectedBeforeEffectsDelivery.run(input, context)
    def compensatedAuthorityCompletion(input: BankRefundOriginal,
        context: CompletionContext): Future[BankRefundResult] =
      bindings.compensatedAuthority.run(input, context)
    def compensatedDeliveryCompletion(input: BankRefundResult, context: CompletionContext): Future[Boolean] =
      bindings.compensatedDelivery.run(input, context)
    def recoveryRequiredAuthorityCompletion(input: BankRefundOriginal,
        context: CompletionContext): Future[BankRefundResult] =
      bindings.recoveryRequiredAuthority.run(input, context)
    def recoveryRequiredDeliveryCompletion(input: BankRefundResult, context: CompletionContext): Future[Boolean] =
      bindings.recoveryRequiredDelivery.run(input, context)
    def outcomeUnknownAuthorityCompletion(input: BankRefundOriginal,
        context: CompletionContext): Future[BankRefundResult] =
      bindings.outcomeUnknownAuthority.run(input, context)
    def outcomeUnknownDeliveryCompletion(input: BankRefundResult, context: CompletionContext): Future[Boolean] =
      bindings.outcomeUnknownDelivery.run(input, context)

    /** Declared identity of every bound implementation, derived from the
      * obligation table — never hand-maintained. */
    lazy val declarations: Map[String, (String, String)] =
      obligations.map { o =>
        val b = o.bound(bindings)
        o.declaredId -> (b.id -> b.version)
      }.toMap
  }

  /** Static binding ownership per port, keyed by declared implementation id:
    * (id, role wire, artifact, version, boundary) — evidence data, derived
    * from the obligation table by the role order above. */
  def bindingOwnership(bindings: Bindings): Vector[(String, String, String, String, String)] = {
    obligations.map { o =>
      val role = o.declaredId match {
        case id if id.endsWith("Entry") => "entry"
        case id if id.endsWith("Association") => "association"
        case id if id.endsWith("Authority") => "authority"
        case id if id.endsWith("Delivery") => "result-delivery"
        case id if id.endsWith("QueryPort") => "phase-query"
        case _ => "phase-execute"
      }
      (o.declaredId, role, "bank-refund-technical", implementationVersion, "handwritten-composition-root")
    }
  }
}
