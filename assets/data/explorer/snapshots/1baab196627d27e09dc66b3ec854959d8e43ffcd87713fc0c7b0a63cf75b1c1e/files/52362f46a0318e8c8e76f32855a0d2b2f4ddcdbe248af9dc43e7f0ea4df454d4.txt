package example.bankwealth.application.connectors

import example.bankwealth.domain.values.{OccurrenceId, OperationId, SipOccurrence}

object BankSipDispatchProtocol {
  final case class BankSipDispatchRequest(occurrenceId: OccurrenceId, occurrence: SipOccurrence)
  final case class BankSipDispatchResult(operationId: OperationId)
  final case class BankSipDispatchOperation(operation: String, request: BankSipDispatchRequest)
  final case class BankSipDispatchObservation(
    operation: String,
    request: BankSipDispatchRequest,
    status: String,
    evidence: String,
    result: Option[BankSipDispatchResult]
  )
  final case class BankSipDispatchReconcile(
    operation: String,
    request: BankSipDispatchRequest,
    decision: String,
    evidence: Option[BankSipDispatchObservation],
    authorization: String
  )
}
