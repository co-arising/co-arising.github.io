/** Persistence-layer protobuf payload codecs for `BankRefundSagaDefinition`. */
object BankRefundProcessCodecs extends BankRefundSagaDefinition.Codecs {

  val argsCodec: PayloadCodec[BankRefundSagaDefinition.Args] = new PayloadCodec[BankRefundSagaDefinition.Args] {
    val pin: CodecPin = BankRefundSagaDefinition.ArgsCodecPin
    def encode(value: BankRefundSagaDefinition.Args): ByteString =
      DomainConverters.encodeBankRefundArgsPO(value).toByteString
    def decode(bytes: ByteString): Try[BankRefundSagaDefinition.Args] =
      Try(DomainConverters.decodeBankRefundArgsPO(
        example.bankwealth.infrastructure.proto.BankRefundArgsPO.parseFrom(bytes.toByteArray)))
  }

    val pin: CodecPin = codecPin
    def encode(value: Receipt): ByteString = encodeReceipt(value)
    def decode(bytes: ByteString): Try[Receipt] = decodeBytes(bytes)
  }

  val DebitFundCollectionPrepareCodec: PayloadCodec[DebitFundCollectionPrepareReceipt] = payloadCodec(
