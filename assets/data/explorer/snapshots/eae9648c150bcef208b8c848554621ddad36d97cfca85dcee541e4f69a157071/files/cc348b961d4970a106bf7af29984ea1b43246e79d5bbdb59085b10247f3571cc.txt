story TaBridgeSettlement layer application

where:
  extern submissionPinned(orderId)          as "该订单已钉住原始 TA 报送凭证"
  extern taReceiptAuthentic(orderId)        as "TA 回执签名与字段白名单校验通过"
  extern taOutcomeConfirmed(orderId)        as "TA 观测结果为确认成功"
  extern taOutcomeRejected(orderId)         as "TA 观测结果为明确拒单"
  extern taAbsenceConfirmed(orderId)        as "TA 对账查询已返回明确缺席证明"

axioms:
  [AX-TA-1] ALWAYS TaBridgeSubmission SHALL NOT supply investorAccountId, fundCollectionAccountId

rules:
  [TA-BR-01] IF not submissionPinned(orderId)
             WHEN BankTaBridgePort.ingestOutcome
             THEN SHALL reject request with TA_UNKNOWN_ORIGINAL

  [TA-BR-02] IF not taReceiptAuthentic(orderId)
             WHILE submissionPinned(orderId)
             WHEN BankTaBridgePort.ingestOutcome
             THEN SHALL reject request with TA_INVALID_RECEIPT

  [TA-BR-03] WHILE submissionPinned(orderId) and taReceiptAuthentic(orderId) and taOutcomeConfirmed(orderId)
             WHEN BankTaBridgePort.ingestOutcome
             SHALL dispatch FundOrder.RecordFundTAConfirmed
               AND SHALL NOT dispatch FundOrder.RequestIndependentRefund
               AND respond TaBridgeAcknowledgement with ack.state == "confirmed"

  [TA-BR-04] WHILE submissionPinned(orderId) and taReceiptAuthentic(orderId)
               and not taOutcomeConfirmed(orderId)
               and (taOutcomeRejected(orderId) or taAbsenceConfirmed(orderId))
             WHEN BankTaBridgePort.ingestOutcome
             SHALL dispatch FundOrder.RequestIndependentRefund
               AND SHALL NOT dispatch FundOrder.RecordFundTAConfirmed
               AND respond TaBridgeAcknowledgement with ack.state == "refund_dispatched"

  [TA-BR-05] WHILE submissionPinned(orderId) and taReceiptAuthentic(orderId)
               and not taOutcomeConfirmed(orderId)
               and not taOutcomeRejected(orderId)
               and not taAbsenceConfirmed(orderId)
             WHEN BankTaBridgePort.ingestOutcome
             SHALL respond TaBridgeAcknowledgement with ack.state == "pending_reconcile"
               AND SHALL NOT dispatch FundOrder.RequestIndependentRefund
               AND SHALL NOT dispatch FundOrder.RecordFundTAConfirmed
